GESTIÓN DEL CURSOR PERSONALIZADOconst cursor = document.getElementById('k-cursor');let mouseX = window.innerWidth / 2, mouseY = window.innerHeight / 2;let ballX = mouseX, ballY = mouseY;const speed = 0.15; // Coeficiente de fricción físicawindow.addEventListener('mousemove', (e) => {mouseX = e.clientX;mouseY = e.clientY;});// Cambios dinámicos del cursor al pasar por elementos interactivosdocument.querySelectorAll('.interactive-element, a, input, button').forEach(el => {el.addEventListener('mouseenter', () => {cursor.style.width = '40px';cursor.style.height = '40px';cursor.style.borderColor = 'var(--accent-magenta)';});el.addEventListener('mouseleave', () => {cursor.style.width = '20px';cursor.style.height = '20px';cursor.style.borderColor = 'var(--accent-cyan)';});});// 2. MOTOR GRAFICO DE INTERFERENCIA CIMÁTICAconst canvas = document.getElementById('canvas-kronos');const ctx = canvas.getContext('2d');let width = canvas.width = window.innerWidth;let height = canvas.height = window.innerHeight;window.addEventListener('resize', () => {width = canvas.width = window.innerWidth;height = canvas.height = window.innerHeight;});// Controles HUD vinculadosconst sliderFreq = document.getElementById('slider-freq');const sliderAmp = document.getElementById('slider-amp');const sliderLines = document.getElementById('slider-lines');const valFreq = document.getElementById('val-freq');const valAmp = document.getElementById('val-amp');const valLines = document.getElementById('val-lines');let lineCount = parseInt(sliderLines.value);let baseFrequencyModifier = parseFloat(sliderFreq.value) / 10000;let baseAmplitude = parseFloat(sliderAmp.value);// Eventos de sliders actualizados en tiempo realsliderFreq.addEventListener('input', (e) => {valFreq.innerText = e.target.value + 'Hz';baseFrequencyModifier = parseFloat(e.target.value) / 10000;updateAudioFreq(parseFloat(e.target.value));});sliderAmp.addEventListener('input', (e) => {valAmp.innerText = e.target.value;baseAmplitude = parseFloat(e.target.value);});sliderLines.addEventListener('input', (e) => {valLines.innerText = e.target.value;lineCount = parseInt(e.target.value);});// 3. MOTOR AUDIO SINTETIZADOR CUÁNTICO (Web Audio API)let audioCtx = null;let mainOscillator = null;let subOscillator = null;let mainGain = null;const engineStatus = document.getElementById('engine-status');function initAudio() {if (audioCtx) return; // Evita duplicar instanciasaudioCtx = new (window.AudioContext || window.webkitAudioContext)();mainGain = audioCtx.createGain();mainGain.gain.setValueAtTime(0.05, audioCtx.currentTime); // Volumen controlado seguromainOscillator = audioCtx.createOscillator();mainOscillator.type = 'sine';mainOscillator.frequency.setValueAtTime(parseFloat(sliderFreq.value) * 2, audioCtx.currentTime);subOscillator = audioCtx.createOscillator();subOscillator.type = 'triangle';subOscillator.frequency.setValueAtTime(parseFloat(sliderFreq.value), audioCtx.currentTime);// Enrutamiento cuánticomainOscillator.connect(mainGain);subOscillator.connect(mainGain);mainGain.connect(audioCtx.destination);mainOscillator.start();subOscillator.start();engineStatus.innerText = 'ONLINE';engineStatus.style.color = 'var(--accent-cyan)';document.getElementById('btn-sync').innerText = 'Sincronía Forzada';}function updateAudioFreq(frequency) {if (!audioCtx) return;mainOscillator.frequency.setTargetAtTime(frequency * 2, audioCtx.currentTime, 0.1);subOscillator.frequency.setTargetAtTime(frequency, audioCtx.currentTime, 0.1);}function modulateAudioWithMouse(xFactor, yFactor) {if (!audioCtx) return;// Modulación espacial: el eje X cambia ligeramente el Pitch Armónico, el eje Y cambia la ganancialet targetFreq = (parseFloat(sliderFreq.value) * 2) + (xFactor * 100);mainOscillator.frequency.setTargetAtTime(targetFreq, audioCtx.currentTime, 0.1);let targetGain = 0.02 + (yFactor * 0.06);mainGain.gain.setTargetAtTime(targetGain, audioCtx.currentTime, 0.1);}// Inicializar audio en clics de la interfazdocument.getElementById('btn-sync').addEventListener('click', initAudio);window.addEventListener('click', () => { if(!audioCtx) initAudio(); });// 4. LOOP PRINCIPAL DE RENDERIZADO (ANIMATION TICK)let phaseOffset = 0;function render() {// Efecto barrido de vectores (Motion blur digital)ctx.fillStyle = 'rgba(2, 3, 8, 0.08)';ctx.fillRect(0, 0, width, height);// Física suave aplicada al cursorballX += (mouseX - ballX) * speed;ballY += (mouseY - ballY) * speed;cursor.style.left = ballX + 'px';cursor.style.top = ballY + 'px';phaseOffset += 0.004;// Modulación del sintetizador de audio basada en coordenadas relativasmodulateAudioWithMouse(ballX / width, 1 - (ballY / height));// Renderizado del entramado de ondasfor (let i = 0; i < lineCount; i++) {let currentLineY = (height / lineCount) * i;let speedFactor = 0.002 + (i * 0.0005);let wavePhase = phaseOffset + (i * 0.15);ctx.beginPath();ctx.lineWidth = (i % 4 === 0) ? 1.5 : 0.6;// Generación de gradiente adaptativo y dinámicolet dynamicGradient = ctx.createLinearGradient(0, 0, width, 0);dynamicGradient.addColorStop(0, 'rgba(168, 85, 247, 0.15)'); // Púrpura de entradadynamicGradient.addColorStop(ballX / width, 'rgba(0, 255, 204, 0.55)'); // Punto de quiebre en el cursordynamicGradient.addColorStop(1, 'rgba(255, 0, 204, 0.15)'); // Magenta de salidactx.strokeStyle = dynamicGradient;for (let x = 0; x < width; x += 15) {// Algoritmo de interferencia cuántica respecto al cursor físicolet distanceToMouse = Math.abs(x - ballX);let influenceArea = Math.max(0, (350 - distanceToMouse) / 350);// Ondas matemáticas puras moduladaslet y = currentLineY + Math.sin(x * baseFrequencyModifier + wavePhase) * (baseAmplitude * 0.4);// Inyección de deformación magnética del mousey += Math.sin(x * 0.015 + wavePhase * 1.8) * (60 * influenceArea) * (1 - Math.abs(ballY - currentLineY)/height);if (x === 0) ctx.moveTo(x, y);else ctx.lineTo(x, y);}ctx.stroke();}requestAnimationFrame(render);}// Encendido del motor gráficorender();
